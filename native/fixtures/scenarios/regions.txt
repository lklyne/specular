# The window's regions hidden and shown by name (ADR 0055): each gives its
# room to the canvas, a tab that shows its item keeps the dock's bar until
# the dock is hidden, the reset key brings back the shipped layout, and
# none of it is written to the canvas.
given page p1 100 100 400 300
given sticky s 700 100 200 200 note
wait 50

# the shipped layout: every region drawn, the tab row across the top and
# the docked tools down the left
expect layout is region toolbar shown
expect layout is insets 36 38 0 0

run ui.toggle {"id":"toolbar"}
expect layout is region toolbar hidden
expect layout is toggled toolbar hidden
expect layout is insets 0 38 0 0
run ui.toggle {"id":"tabs"}
expect layout is insets 0 0 0 0

# the sidebar takes room only while it is open and the layout draws it
sidebar on
expect layout is insets 256 0 0 0
run ui.hide {"id":"sidebar"}
expect sidebar on
expect layout is region sidebar hidden
expect layout is insets 0 0 0 0
run ui.show {"id":"sidebar"}
expect layout is insets 256 0 0 0
expect layout lacks toggled sidebar
sidebar off

# a tab showing its item keeps the dock's bar under the tab row
run ui.show {"id":"tabs"}
select p1
control view.open
expect showing p1
expect layout is insets 0 74 0 0
run ui.toggle {"id":"dock"}
expect layout is insets 0 38 0 0
key escape
control view.canvas
expect showing canvas

# a control is toggled by its id like a region
run ui.hide {"id":"tool.inspect"}
expect layout is toggled tool.inspect hidden

# the reset key sets every toggle aside
key ctrl+alt+shift+escape
expect layout lacks toggled
expect layout is region toolbar shown
expect layout is region dock shown
expect layout is insets 36 38 0 0

# and so does ui.reset
run ui.toggle {"id":"toolbar"}
run ui.reset
expect layout is insets 36 38 0 0
expect saved same-as start
undo-to-start
