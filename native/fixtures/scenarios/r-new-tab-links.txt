# canvas: anchoring.canvas
# Links a page opens in a new tab: a target=_blank click (foreground), a
# Command-click (background) and a javascript: link, which opens nothing.
# Each new page lands beside the source at its size, joins no sync set and
# is one undo step.
camera 0,0,0.5
save 01-start.canvas
page-opens-tab p1 https://example.com/opened
save 02-foreground.canvas
snapshot 02-foreground.png
page-opens-tab p1 https://example.com/behind background
save 03-background.canvas
page-opens-tab p1 javascript:void(0)
save 04-refused.canvas
key escape
key cmd+z
key cmd+z
save 05-undone.canvas
