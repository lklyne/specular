# canvas: layout.canvas
# Lines (ADR 0015): a managed row laid out again by whatever changes it,
# its reorder dots and gap strips, and the same handles on a loose even
# selection with no group. `g` is a managed row of `a`, `b` and `c`, 100
# wide and 40 apart, with `d` loose below it; `la`, `lb` and `lc` are a
# loose even row.
camera 0,0,1
expect saved same-as layout.canvas

# a line shows its handles while it or a member is selected, and not with
# a tool armed
expect dots none
select g
expect dots a b c
expect gap-strips 2
select b
expect dots a b c
select a c
expect dots a b c
tool text
expect dots none
key escape
key escape
# a dot is over its box and a strip fills the gap; off the dot, the box is
# a body again
select g
expect hit 150 150 dot a
expect hit 220 150 gap x
expect hit 156 144 dot a
expect hit 120 120 body a
expect hit 158 150 body a

# whatever changes a managed row lays it out again in the same step: its
# gap set from the dock
control group.gap.inc
expect saved /nodes/g/layoutGap = 60
expect entity b at 260 100
control group.gap.dec
expect entity b at 240 100
key cmd+z
key cmd+z
# a member resized by its right handle
select a
drag 200 150 260 150
expect entity b at 300 100
expect entity c at 440 100
expect entity g rect 76 76 488 148
key cmd+z
# a member deleted
select a
key delete
expect entity b at 240 100
expect entity g rect 216 76 288 148
key cmd+z
# a member sent to the back leads
select c
key cmd+shift+[
expect entity c at 100 100
expect entity a at 240 100
key cmd+z
# turned into a column, and turned off from the dock
select g
control group.layout.column
expect saved /nodes/g/layoutMode = column
expect entity c at 100 380
expect entity g rect 76 76 148 428
control group.layout.column
expect saved /nodes/g/managedLayout = false
expect shown-control group.gap.inc no
key cmd+z
key cmd+z
expect saved same-as start
# an item dropped into it joins last, from in front of everything
key escape
drag 130 530 270 170
expect entity d parent g
expect entity d rect 520 100 60 60
expect entity g rect 76 76 528 148
key cmd+z
# one sent behind everything first joins first
select d
key cmd+shift+[
key escape
drag 130 530 270 170
expect entity d rect 100 100 60 60
key cmd+z
key cmd+z
# dragging a member's body, off its dot, moves the whole group
select b
drag 310 120 350 220
expect entity a at 140 200
expect entity b at 280 200
expect entity g rect 116 176 428 148
expect entity b parent g
key cmd+z
# turned off, its members stay and then move freely
select g
control group.layout.row
expect saved /nodes/g/managedLayout = false
expect saved /nodes/g/layoutMode = freeform
expect entity b at 240 100
select a
drag 150 150 150 250
expect entity a at 100 200
expect entity b at 240 100
key cmd+z
key cmd+z
expect saved same-as start

# dragging c's dot to the front of the row moves it there: in flight the
