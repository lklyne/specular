# canvas: ../../pages.canvas
# Real pages: a comment on a page's element, a region over a button, the
# tall page entered and scrolled, and a reload. The tall page is at screen
# x 715..1315, y 80..455 with this camera, in a 1400x800 viewport.
camera 40,80,0.75
wait 500
# A comment on the heading: the page names the element.
tool comment
click 760 150
wait 300
type On the heading
key enter
expect count comments 1
expect saved /annotations/0/anchor/type = element
expect saved /annotations/0/anchor/pageId = p-tall
# A region over the button grabs it, so the region belongs to the page.
drag 730 200 870 260
wait 300
type Region over the button
key enter
expect count comments 2
expect saved /annotations/1/pageAnchor/pageId = p-tall
tool select
snapshot 02-commented.png
# Enter the tall page and scroll it: the region and the pin follow.
click 1000 400
click 1000 400
expect focus page p-tall
wheel 0 -150
wait 500
snapshot 03-scrolled.png
key escape
# In-place navigation.
select p-example
act page-reload
wait 1000
expect entity p-example text https://example.com/
undo-to-start
