# canvas: anchoring.canvas
# macOS Reduce motion: with transitions on as in a window, opening a page's
# tab springs the camera, camera input ends the spring back to the canvas
# where it is, and a page turned on its side turns its shell by a spring
# while its rect lands at once; with Reduce motion on, each lands at once.
# The camera's first frame is pinned in specular-interact's
# tests/it/motion.rs. Then the settings' cursor switches, which change no
# document.
camera 0,0,0.5
motion on
select p1
control view.open
expect showing p1
# with transitions on, the camera has not moved yet
expect camera 0,0,0.5
expect motion running
# the spring's clock starts at the first frame after the change
wait 10
wait 2000
expect motion still
control view.canvas
expect motion running
wait 10
wait 60
# a scroll ends the way back where the camera is
wheel 0 40
expect motion still
wait 2000
camera 0,0,0.5
# a page turned on its side: its rect at once, its shell by a spring, and
# an undo is not a turn
select p1
act landscape
expect entity p1 size 1280 800
expect motion running
key cmd+z
wait 10
expect motion still
expect entity p1 size 480 360
reduce-motion on
select p1
act landscape
expect motion still
key cmd+z
select p1
control view.open
# with Reduce motion on, it is there at once
expect camera 478,257,1
control settings.agent-cursors
control settings.synced-cursors
expect saved same-as start
