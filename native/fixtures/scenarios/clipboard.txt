# canvas: empty.canvas
# Copy and paste: text between two stickies, a whole entity, a URL, and
# plain text with nothing selected.
camera 0,0,1

tool sticky
click 200 160
type alpha beta gamma
key escape
tool sticky
click 520 160
type second
key escape

# copy the first sticky's text, paste it into the second
double-click 300 260
key cmd+c
key escape
double-click 620 260
key cmd+down
key enter
key cmd+v
expect editing second\nalpha beta gamma
key escape

# cut a word out of the first
double-click 300 260
key cmd+up
hold shift+alt
key right
hold none
key cmd+x
expect editing \sbeta gamma
key escape

# copy the first sticky itself and paste it at the pointer
click 300 300
key cmd+c
move 900 500
key cmd+v
expect count sticky 3
expect entity @top rect 900 500 200 200
expect entity @top text \sbeta gamma
expect selection @top

# a URL from another app, with nothing selected, becomes a page
key escape
clipboard https://example.com/pricing
move 300 620
key cmd+v
wait 200
expect entity @top kind page
expect entity @top text https://example.com/pricing

# plain text with nothing selected becomes a sticky
key escape
clipboard Just a sentence copied from somewhere else.\nWith a second line.
move 1200 200
key cmd+v
expect entity @top kind sticky
expect entity @top rect 1200 200 200 200
expect entity @top text Just a sentence copied from somewhere else.\nWith a second line.
expect count entities 5
undo-to-start
