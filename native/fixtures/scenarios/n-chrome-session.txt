# canvas: ../kitchen-sink.canvas
# One session through all of the built-in chrome with the sidebar shown: a
# page picked from its row, the dock's address and size list, the
# right-click menu on a sticky (a duplicate), a second canvas
# added and a sticky placed on it, the first canvas switched back to, and
# the first canvas saved. The window is 1600x1000, the chrome the top 118
# pixels and the sidebar the left 256; canvas rows are 28 pixels, the first
# centred at y 170.
camera -2300,-1600,1
sidebar on
save 00-start.canvas

# the list scrolled to its end, where a page's row is; the row brings it into
# view and its controls fill the dock
move 100 500
wheel 0 -5000
control sidebar.pages.page-desktop
snapshot 01-page-from-its-row.png
control page.url
key cmd+a
type example.org/docs
key enter
snapshot 02-address-entered.png
control page.size
snapshot 03-size-list.png
key escape

# a sticky's menu opens beside the sidebar and duplicates
camera 200,-400,1
right-click 500 250
snapshot 04-menu-sticky.png
control menu.duplicate
save 05-duplicated.canvas
snapshot 05-duplicated.png

# a new canvas takes a sticky of its own
control sidebar.add
control tool.sticky
click 700 400
type second canvas
key escape
save 06-new-canvas.canvas
snapshot 06-new-canvas.png

# back on the first canvas nothing of the second is there
click 100 170
snapshot 07-first-canvas-again.png
save 07-first-canvas.canvas
