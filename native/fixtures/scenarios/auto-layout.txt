# canvas: guides.canvas
# Auto-layout: three shapes made a managed row, its dots and gap bars, the
# popup's gap and direction, a gap strip dragged, a dot dragged to another
# slot with the box floating under the pointer, a loose even selection
# reordered with no group, and every step undone.
camera 0,0,1
save 00-start.canvas

# three shapes become a row, in the order they read, 80 apart
select left right a

control group.gap.dec
snapshot 02-gap-60.png
control group.layout.column
snapshot 03-column.png
control group.layout.row
