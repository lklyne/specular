# canvas: ../kitchen-sink.canvas
# Zoom to both ends, then pan and zoom while a gesture is in flight: the
# thing under the pointer should stay under the pointer.
camera fit

# pinch out as far as it goes, about the middle
move 800 500
pinch -1
pinch -1
pinch -1
pinch -1
pinch -1
pinch -1
pinch -1
pinch -1
expect zoom 0.02

# and in as far as it goes, over the stickies
camera fit
move 520 200
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
pinch 1
expect zoom 3

# wheel-zoom with the command key
camera 32,-368,0.8
move 112 320
hold cmd
wheel 0 120
wheel 0 120
hold none
expect zoom 1.28

release

expect undo some
key cmd+z
