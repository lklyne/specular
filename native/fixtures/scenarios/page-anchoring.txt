# canvas: anchoring.canvas
# Page anchoring in use: a sticky and a text hooked to a page, an edge from a
# free shape to the sticky, and a region comment in the page's document. The
# page is scrolled, and the hooked items are resized, edited, duplicated and
# freed while it has carried them. The camera puts canvas (0, 0) at screen
# (60, 60), so the page is at (160, 160) to (640, 520).
camera 60,60,1
expect saved same-as anchoring.canvas

# enter the page and scroll it 60 down: the sticky and the text go 60 up
# with its content, and the edge still meets the sticky
click 620 500
click 620 500
wheel 0 -60
wait 100
expect focus page p1

# 80 more and the sticky is half out of the page's top: it fades over the
# 48 pixels above the edge and is cut at the page's sides
wheel 0 -80
wait 100
snapshot 02-fading-at-the-top.png
wheel 0 80
wait 100
key escape
key escape

# selected where it is seen, and resized from its bottom-right handle
click 300 240
expect selection note
drag 380 300 420 340
# the page names the cell under it in its own time
wait 100
# resized with the page 60 down: stored where it was seen, restamped
# there, and attached to the grid cell under its middle, in Electron's shape
expect saved /nodes/note/x = 160
expect saved /nodes/note/y = 120
expect saved /nodes/note/width = 200
expect saved /nodes/note/height = 200
expect saved /nodes/note/specular/pageAnchor/scrollY = 60
expect saved /nodes/note/specular/pageAnchor/element keys selector docX docY

# the text edited while shifted: the caret is at the end of what is drawn
key escape
double-click 440 350
key cmd+right
type , edited
key shift+cmd+left
key escape
key escape
wait 100

# a region comment over two cells of the page's grid is the page's
tool comment
press 180 400
drag-to 400 450
drag-to 500 500
release
wait 100
type Tighten this row
key enter
wait 100
# a region of the page is attached once, in the page's document space
expect count comments 1
expect saved /annotations/0/anchor keys type docRect
expect saved /annotations/0/pageAnchor/element keys selector docX docY
tool select
key escape

# scrolled 40 more, the region, the sticky and the text all follow
click 620 500
click 620 500
wheel 0 -40
wait 100
key escape
key escape

# a duplicate of the sticky lands beside the page and is free; the page is
# then deleted, which frees the sticky and the text where they are seen
select note
key cmd+d
camera 60,60,1
# a text edited while carried is attached where it is seen
expect saved /nodes/caption/specular/pageAnchor/element/docY = 240
# a duplicate beside the page is free
expect saved /nodes/@top/specular/pageAnchor absent
expect saved /nodes/@top/y = 80
mark duplicated
select p1
key backspace
expect sent close-page p1
# deleting the page frees what was hooked to it where it was seen, 100
# above where the sticky was stored
expect entity p1 gone
expect saved /nodes/note/specular/pageAnchor absent
expect saved /nodes/caption/specular/pageAnchor absent
expect saved /nodes/note/y = 80
expect saved /nodes/caption/y = 240
expect saved /annotations/0/pageAnchor absent
key cmd+z
expect sent create-page p1
expect saved same-as mark duplicated
# the cell the page named for the edited caption is not taken back by
# undo, which restores its text
undo-to-start except /nodes/caption/specular/pageAnchor/element
